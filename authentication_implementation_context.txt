Absolutely. This is a great security feature to add to EduGen AI. You can integrate Google Authenticator-based Two-Factor Authentication (2FA) without using any Google API or paying for an authentication service.

The technology is called TOTP (Time-Based One-Time Password). Google Authenticator generates a six-digit code using a shared secret and the current time. The code typically changes every 30 seconds. Your backend independently calculates the expected code and checks whether it matches.

Step C — Encrypt the authenticator secret
Never store the authenticator secret as plain text in PostgreSQL.

Create:

backend/src/utils/totpCrypto.js
const crypto = require('crypto');

function getKey() {
  const key = Buffer.from(process.env.TOTP_ENCRYPTION_KEY || '', 'hex');

  if (key.length !== 32) {
    throw new Error('Invalid TOTP encryption key');
  }

  return key;
}

function encryptSecret(secret) {
  const iv = crypto.randomBytes(12);
  const cipher = crypto.createCipheriv('aes-256-gcm', getKey(), iv);

  const encrypted = Buffer.concat([
    cipher.update(secret, 'utf8'),
    cipher.final()
  ]);

  return [
    iv.toString('hex'),
    cipher.getAuthTag().toString('hex'),
    encrypted.toString('hex')
  ].join(':');
}

function decryptSecret(value) {
  const [iv, tag, data] = value.split(':');

  const decipher = crypto.createDecipheriv(
    'aes-256-gcm',
    getKey(),
    Buffer.from(iv, 'hex')
  );

  decipher.setAuthTag(Buffer.from(tag, 'hex'));

  return Buffer.concat([
    decipher.update(Buffer.from(data, 'hex')),
    decipher.final()
  ]).toString('utf8');
}

module.exports = { encryptSecret, decryptSecret };


Generate an encryption key using:
node -e "console.log(require('crypto').randomBytes(32).toString('hex'))"


Add the generated value to your backend .env:
TOTP_ENCRYPTION_KEY=YOUR_GENERATED_64_CHARACTER_HEX_KEY


Keep this key private, never commit it to GitHub, and back it up securely. Losing it means you cannot decrypt existing authenticator secrets.

1. Generate the QR code
async function setupTotp(req, res) {
  try {
    const user = await User.findById(req.user.id);

    if (!user) return res.status(404).json({ error: 'User not found' });
    if (user.totpEnabled) {
      return res.status(409).json({ error: 'Authenticator already enabled' });
    }

    const { generateSecret, generateURI } = await import('otplib');

    const secret = generateSecret();

    const uri = generateURI({
      issuer: 'EduGen AI',
      label: user.email,
      secret
    });

    const qrCode = await QRCode.toDataURL(uri);

    await User.savePendingTotpSecret(
      user.id,
      encryptSecret(secret)
    );

    res.json({ qrCode });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Unable to generate QR code' });
  }
}


2. Verify and activate the authenticator
async function enableTotp(req, res) {
  try {
    const { code } = req.body;

    if (typeof code !== 'string' || !/^\d{6}$/.test(code)) {
      return res.status(400).json({ error: 'Enter a valid six-digit code' });
    }

    const user = await User.findById(req.user.id);

    if (!user?.totpPendingSecretEncrypted) {
      return res.status(400).json({ error: 'Setup authenticator first' });
    }

    const { verify } = await import('otplib');

    const secret = decryptSecret(user.totpPendingSecretEncrypted);

    const result = await verify({
      secret,
      token: code,
      epochTolerance: 30
    });

    if (!result.valid) {
      return res.status(400).json({ error: 'Invalid verification code' });
    }

    await User.enableTotp(
      user.id,
      user.totpPendingSecretEncrypted
    );

    res.json({ message: 'Two-factor authentication enabled' });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Unable to enable authentication' });
  }
}

3. Verify the login code

Add this controller function:
async function verifyTotpLogin(req, res) {
  try {
    const { challengeToken, code } = req.body;

    if (
      typeof code !== 'string' ||
      !/^\d{6}$/.test(code) ||
      typeof challengeToken !== 'string'
    ) {
      return res.status(400).json({ error: 'Invalid verification request' });
    }

    const payload = jwt.verify(
      challengeToken,
      process.env.JWT_SECRET
    );

    if (payload.purpose !== 'totp_login') {
      return res.status(401).json({ error: 'Invalid login challenge' });
    }

    const user = await User.findById(payload.id);

    if (!user?.totpEnabled || !user.totpSecretEncrypted) {
      return res.status(401).json({ error: 'Authentication failed' });
    }

    const { verify } = await import('otplib');

    const secret = decryptSecret(user.totpSecretEncrypted);

    const result = await verify({
      secret,
      token: code,
      epochTolerance: 30
    });

    if (!result.valid) {
      return res.status(401).json({ error: 'Invalid authentication code' });
    }

    return res.json({
      token: signToken(user),
      user: toUserDTO(user)
    });
  } catch (err) {
    return res.status(401).json({ error: 'Invalid or expired login challenge' });
  }
}

6. Important security considerations
Before treating this as production-ready, address these points:
1. Rate limiting: Restrict password and OTP attempts to prevent brute-force guessing.
2. Secret encryption: Use AES-256-GCM and keep the encryption key outside PostgreSQL.
3. Token separation: Never use the temporary MFA challenge token to access dashboard APIs.
4. Replay protection: Track the last accepted TOTP time step and prevent the same code being accepted more than once.
5. Recovery: Provide backup recovery codes for users who lose their phones.
6. HTTPS: Use HTTPS when deploying. Avoid transmitting authentication data over unencrypted connections.
The examples above establish the basic MFA flow; replay prevention, single-use challenge tracking and recovery codes should be added before production deployment.
